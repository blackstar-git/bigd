1. CREATING VECTORS
-------------------
# Using c()
data1 <- c(10, 20, 30, 40, 50)

x <- seq(from = 2, to = 10, by = 2)
print(x)

# Equally spaced values
y <- seq(from = 0, to = 1, length.out = 5)
print(y)

# Using colon operator

print(z)

# Descending sequence
z <- 10:1

3. COMBINING DATASETS
---------------------
A <- c(6, 7, 8, 9)
B <- c(1, 2, 4, 5)

# Combine into one vector
C <- c(A, B)
print(C)

# Combine as columns
print(cbind(A, B))

# Combine as rows
print(rbind(A, B))

# Merge two data frames using a common ID
employees <- data.frame(
  ID = c(1, 2, 3),
  Name = c("Amit", "Riya", "John")
)

salary <- data.frame(
  ID = c(1, 2, 3),
  Salary = c(25000, 30000, 28000)
)

result <- merge(employees, salary, by = "ID")
print(result)

# Add new employee rows using rbind()
existing <- data.frame(
  ID = c(1, 2),
  Name = c("Amit", "Riya"),
  Salary = c(25000, 30000)
)

new_employees <- data.frame(
  ID = c(3, 4),
  Name = c("John", "Sara"),
  Salary = c(28000, 32000)
)

existing <- rbind(existing, new_employees)
print(existing)


4. SUBSETTING DATA
------------------
students <- data.frame(
  Name = c("Amit", "Riya", "John", "Sara", "Raj"),
  Marks = c(75, 45, 90, 60, 35),
  Attendance = c(85, 70, 95, 80, 60)
)

# Extract rows and columns using [ ]
students[1, ]                              # First row
students[, 1]                              # First column
students[1:3, ]                            # First three rows
students[, c("Name", "Marks")]             # Selected columns
students[students$Marks < 60, ]            # Rows where marks < 60

# Extract a column using [[ ]] and $
students[["Marks"]]
students$Marks

# Filter using subset()
subset(students, Marks < 60)
subset(students, Marks > 60)
subset(students, Marks > 60 & Attendance > 80)
subset(students, Marks > 80 | Attendance > 90)
subset(students, Marks >= 60, select = c(Name, Marks))

# Logical operators:
# <  >  <=  >=  ==  !=  &  |  !


5. SORTING FUNCTIONS
--------------------
x <- c(40, 10, 30, 50, 20)

sort(x)                              # Ascending
sort(x, decreasing = TRUE)           # Descending
order(x)                             # Positions in ascending order
x[order(x)]                          # Sort using order()
x[order(x, decreasing = TRUE)]       # Descending using order()
rev(x)                               # Reverse original order
rev(sort(x))                         # Sort descending using rev()

8. DATA STRUCTURES
------------------
# Vector: one-dimensional, same basic type
v <- c(10, 20, 30, 40)
print(v)

# List: can contain different data types
my_list <- list("Amit", 20, TRUE)
print(my_list)
my_list[[1]]                            # Access first element

# Matrix: two-dimensional, same basic type
m <- matrix(c(1, 2, 3, 4, 5, 6),
            nrow = 2, ncol = 3)
print(m)
m[1, 2]                                 # First row, second column

# Array: multiple dimensions, same basic type
a <- array(1:8, dim = c(2, 2, 2))
print(a)

# Factor: categorical data
grade <- factor(c("A", "B", "A", "C"))
print(grade)
levels(grade)

# Data frame: columns can have different data types
students_df <- data.frame(
  Name = c("Amit", "Riya"),
  Marks = c(75, 85)
)
print(students_df)


9. FUNCTIONS, SCRIPTS, CONDITIONS AND LOOPS

# User-defined function
add <- function(a, b) {
  result <- a + b
  return(result)
}

print(add(10, 20))
print(add(5, 15))

# If-else statement
marks <- 75
if (marks >= 40) {
  print("Pass")
} else {
  print("Fail")
}

# For loop
for (i in 1:5) {
  print(i)
}

# While loop
i <- 1
while (i <= 5) {
  print(i)
  i <- i + 1
}
